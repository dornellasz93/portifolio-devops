def calcular_situacao_academica(notas):
    """Calcula a média e determina a situação do estudante"""
    if not notas:
        return "Sem notas registradas"
    
    media = sum(notas) / len(notas)
    situacao = "Aprovado" if media >= 6.0 else "Exame Final"
    
    return {
        "aluno": "João Marcelo Bergamini",
        "matricula": "2026101202",
        "media": round(media, 2),
        "status": situacao
    }

resultado = calcular_situacao_academica([8.0, 7.5, 9.0])
print(resultado)
